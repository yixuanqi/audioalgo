cmake_minimum_required(VERSION 3.28)
project(AudioAlgoExternalNode LANGUAGES CXX)
find_package(AudioAlgoNodeSDK 1 CONFIG REQUIRED)
add_library(external_stateful SHARED plugin.cpp)
target_link_libraries(external_stateful PRIVATE AudioAlgo::node_abi)
target_compile_features(external_stateful PRIVATE cxx_std_20)
set(PLUGIN_REVISION "1" CACHE STRING "Example behavior revision (1 or 2)")
configure_file(manifest.json.in manifest.json @ONLY)
file(READ "${CMAKE_CURRENT_BINARY_DIR}/manifest.json" PLUGIN_MANIFEST)
configure_file(descriptor.h.in descriptor.h @ONLY)
target_include_directories(external_stateful PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(external_stateful PRIVATE PLUGIN_REVISION=${PLUGIN_REVISION})
set_target_properties(external_stateful PROPERTIES PREFIX "" OUTPUT_NAME "external_stateful")
install(TARGETS external_stateful RUNTIME DESTINATION . LIBRARY DESTINATION .)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/manifest.json" DESTINATION .)
