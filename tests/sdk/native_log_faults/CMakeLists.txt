cmake_minimum_required(VERSION 3.24)
project(AudioAlgoNativeLogFaults LANGUAGES CXX)
find_package(AudioAlgoNodeSDK CONFIG REQUIRED)
configure_file(manifest.json.in manifest.json @ONLY)
file(READ "${CMAKE_CURRENT_BINARY_DIR}/manifest.json" MANIFEST_JSON)
configure_file(descriptor.h.in descriptor.h @ONLY)
add_library(native_log_faults SHARED plugin.cpp)
target_link_libraries(native_log_faults PRIVATE AudioAlgo::node_abi)
target_compile_features(native_log_faults PRIVATE cxx_std_20)
target_compile_definitions(native_log_faults PRIVATE LOG_MODE=0)
target_include_directories(native_log_faults PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
set_target_properties(native_log_faults PROPERTIES OUTPUT_NAME native_logging RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_BINARY_DIR}" ARCHIVE_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_BINARY_DIR}")

add_executable(native_log_fault_consumer consumer.cpp)
target_link_libraries(native_log_fault_consumer PRIVATE AudioAlgo::node_abi)
target_compile_features(native_log_fault_consumer PRIVATE cxx_std_20)
enable_testing()
add_test(NAME native_log_fault_status COMMAND native_log_fault_consumer "$<TARGET_FILE:native_log_faults>")
