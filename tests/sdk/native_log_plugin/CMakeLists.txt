cmake_minimum_required(VERSION 3.24)
project(AudioAlgoOptionalLogConsumer LANGUAGES CXX)
find_package(AudioAlgoNodeSDK CONFIG REQUIRED)
foreach(mode good size version function rejected absent)
  if(mode STREQUAL "good")
    set(LOG_MODE 0)
  elseif(mode STREQUAL "size")
    set(LOG_MODE 1)
  elseif(mode STREQUAL "version")
    set(LOG_MODE 2)
  elseif(mode STREQUAL "function")
    set(LOG_MODE 3)
  elseif(mode STREQUAL "rejected")
    set(LOG_MODE 4)
  else()
    set(LOG_MODE 5)
  endif()
  file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/${mode}")
  configure_file(manifest.json.in "${mode}/manifest.json" @ONLY)
  file(READ "${CMAKE_CURRENT_BINARY_DIR}/${mode}/manifest.json" MANIFEST_JSON)
  configure_file(descriptor.h.in "${mode}/descriptor.h" @ONLY)
  add_library(log_${mode} SHARED plugin.cpp)
  target_link_libraries(log_${mode} PRIVATE AudioAlgo::node_abi)
  target_compile_features(log_${mode} PRIVATE cxx_std_20)
  target_compile_definitions(log_${mode} PRIVATE LOG_MODE=${LOG_MODE})
  target_include_directories(log_${mode} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/${mode}")
  set_target_properties(log_${mode} PROPERTIES OUTPUT_NAME native_logging
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/${mode}"
    RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_BINARY_DIR}/${mode}"
    ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/${mode}"
    ARCHIVE_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_BINARY_DIR}/${mode}")
endforeach()
