cmake_minimum_required(VERSION 3.24)
project(AudioAlgoNativeLogDetach LANGUAGES CXX)
find_package(AudioAlgoNodeSDK CONFIG REQUIRED)
configure_file(manifest.json.in manifest.json @ONLY)
file(READ "${CMAKE_CURRENT_BINARY_DIR}/manifest.json" MANIFEST_JSON)
configure_file(descriptor.h.in descriptor.h @ONLY)
add_library(native_log_detach SHARED plugin.cpp)
target_link_libraries(native_log_detach PRIVATE AudioAlgo::node_abi)
target_compile_features(native_log_detach PRIVATE cxx_std_20)
target_compile_definitions(native_log_detach PRIVATE LOG_MODE=0)
target_include_directories(native_log_detach PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
set_target_properties(native_log_detach PROPERTIES OUTPUT_NAME native_logging RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_BINARY_DIR}" ARCHIVE_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_BINARY_DIR}")
