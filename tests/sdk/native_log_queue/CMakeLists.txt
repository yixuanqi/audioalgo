cmake_minimum_required(VERSION 3.24)
project(AudioAlgoLogQueueConsumer LANGUAGES C CXX)
find_package(AudioAlgoNodeSDK CONFIG REQUIRED)
add_executable(native_log_queue_consumer consumer.cpp abi_check.c)
target_link_libraries(native_log_queue_consumer PRIVATE AudioAlgo::node_abi)
target_compile_features(native_log_queue_consumer PRIVATE c_std_11 cxx_std_20)
if(MSVC)

endif()
enable_testing()
add_test(NAME native_log_queue_contract COMMAND native_log_queue_consumer)
